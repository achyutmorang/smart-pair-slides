\section{Phase II Baseline-First Framework}

\begin{frame}{Pivot Strategy}
\begin{itemize}
    \item Move from objective-first novelty to benchmark-first rigor.
    \item Reproduce strong WOSAC-aligned baselines before new signal claims.
    \item Keep surprise as gated auxiliary signal, not default primary objective.
\end{itemize}
\end{frame}

\begin{frame}{Protocol in Phase II}
\begin{itemize}
    \item Fixed dataset splits, seed maps, and compute budgets.
    \item Matched-budget comparisons: random, risk-only, surprise variants, joint.
    \item Mandatory artifact exports:
    \begin{itemize}
        \item per-scenario metrics
        \item calibration manifests
        \item fairness and trace diagnostics
    \end{itemize}
    \item Statistical checks via repeated seeds and bootstrap/permutation style tests.
\end{itemize}
\end{frame}
